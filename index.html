// 多資料庫極速查詢與本地快取 const isbnCache = new Map(); const fetchBookByISBN = async (isbn) => { const cleanIsbn = isbn.replace(/[^0-9X]/gi, ''); if (!cleanIsbn) return; // 1. 命中快取直接秒出 if (isbnCache.has(cleanIsbn)) { const cached = isbnCache.get(cleanIsbn); newBook.value.title = cached.title; newBook.value.author = cached.author; newBook.value.publisher = cached.publisher; alert(`✓ [快取秒開] 成功帶入書籍：${cached.title}`); return; } // 2. 優先向 Google Books 請求（設置 2 秒超時） const fetchGoogle = async () => { const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), 2000); const res = await fetch(`https://www.googleapis.com/books/v1/volumes?q=isbn:${cleanIsbn}`, { signal: controller.signal }); clearTimeout(timeoutId); const data = await res.json(); if (data.totalItems > 0) { const info = data.items[0].volumeInfo; return { title: info.title || '', author: (info.authors && info.authors.join(', ')) || '', publisher: info.publisher || '' }; } throw new Error('Google Books 查無資料'); }; // 3. 備援向 Open Library 請求 const fetchOpenLibrary = async () => { const res = await fetch(`https://openlibrary.org/api/books?bibkeys=ISBN:${cleanIsbn}&jscmd=data&format=json`); const data = await res.json(); const bookData = data[`ISBN:${cleanIsbn}`]; if (bookData) { return { title: bookData.title || '', author: (bookData.authors && bookData.authors.map(a => a.name).join(', ')) || '', publisher: (bookData.publishers && bookData.publishers[0]?.name) || '' }; } throw new Error('Open Library 查無資料'); }; try { let result = null; try { result = await fetchGoogle(); } catch (e) { // Google 逾時或無資料時切換備援庫 result = await fetchOpenLibrary(); } if (result && result.title) { newBook.value.title = result.title; newBook.value.author = result.author; newBook.value.publisher = result.publisher; isbnCache.set(cleanIsbn, result); alert(`✓ 成功帶入書籍：${result.title}`); } else { alert(`已辨識 ISBN：${cleanIsbn}，線上資料庫暫無收錄，請手動補齊書名。`); } } catch (err) { alert(`已辨識 ISBN：${cleanIsbn}，線上資料庫暫無收錄，請手動輸入書名。`); } };
